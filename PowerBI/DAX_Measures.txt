Total Revenue = SUM(Revenue_Summary[revenue_amount])
Average Satisfaction = AVERAGE(Patient_Feedback[rating])
Total Diagnoses = DISTINCTCOUNT(Diagnoses[diagnosis_id])
Total Lab Tests = DISTINCTCOUNT(Laboratory_Tests[test_id])

Total Admissions = DISTINCTCOUNT(Admissions[admission_id])
Total Beds = COUNTROWS(Beds)
Available Beds = CALCULATE(COUNTROWS(Beds), Beds[status] = "Available")
Bed Availability Rate = DIVIDE([Available Beds], [Total Beds])
Approved Claims = CALCULATE(COUNTROWS(Insurance_Claims), Insurance_Claims[status] = "Approved")
Adjudicated Claims = CALCULATE(COUNTROWS(Insurance_Claims), Insurance_Claims[status] IN {"Approved","Rejected"})
Claim Approval Rate = DIVIDE([Approved Claims], [Adjudicated Claims])
